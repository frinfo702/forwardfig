Transformer

Vaswani et al., Attention Is All You Need (2017) · Fig. 1
flow BT
group enc repeat="N×" {
  e_add op "+"
  e_mha mha
  e_n1  norm
  e_ff  ffn
  e_n2  norm
}
group dec repeat="N×" {
  d_add  op "+"
  d_mmha mha "Masked\\Multi-Head Attention"
  d_n1   norm
  d_mha  mha
  d_n2   norm
  d_ff   ffn
  d_n3   norm
}
x    text "Inputs"
y    text "Outputs\\(shifted right)"
inp  embed "Input Embedding"
out  embed "Output Embedding"
lin  linear
sm   softmax
prob text "Output Probabilities"
e_pe op "~" left-of=e_add
d_pe op "~" right-of=d_add

x -> inp -> e_add -> e_mha -> e_n1 -> e_ff -> e_n2
y -> out -> d_add -> d_mmha -> d_n1 -> d_mha -> d_n2 -> d_ff -> d_n3 -> lin -> sm -> prob
e_add -> e_n1
e_n1 -> e_n2
d_add -> d_n1
d_n1 -> d_n2
d_n2 -> d_n3
e_n2 -> d_mha
e_pe -> e_add
d_pe -> d_add
